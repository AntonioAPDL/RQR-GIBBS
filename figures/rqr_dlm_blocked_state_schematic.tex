  \node[box, text width=3.75cm] (prior) {
    \textbf{Stacked Gaussian state prior}\par\smallskip
    \(\alpha_t=(\theta_{1t}^{\top},\theta_{2t}^{\top})^{\top}\)

    \(\mathcal W_t\).
  };

  \node[warning, text width=4.05cm, right=6mm of prior] (quartic) {
    \textbf{Quartic joint obstruction}\par\smallskip
    The squared augmented pseudo-residual contains
    \(\eta_{1t}^{2}\eta_{2t}^{2}\). The joint log posterior is
    therefore not Gaussian, so one ordinary simultaneous FFBS draw
    is unavailable.
  };

  \node[box, text width=4.05cm, right=6mm of quartic] (blocked) {
    \textbf{Condition one path, then FFBS}\par\smallskip
    root 1 given root 2 \(\longrightarrow\)\par
    root 2 given new root 1 \(\longrightarrow\)\par
    optional global label swap.\par\smallskip
    Each path is linear Gaussian after conditioning on the other.
  };

  \draw[arrow] (prior) -- (quartic);
  \draw[arrow] (quartic) -- (blocked);
